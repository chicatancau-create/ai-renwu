def gitignore_excerpt(cap=22):
    out, seen = [], set()
    for i, ln in enumerate(lines_of(".gitignore"), 1):
        if any(k in ln for k in GITIGNORE_KEYS) and i not in seen:
            out.append("%d: %s" % (i, ln)); seen.add(i)
        if len(out) >= cap:
